<script>
  // --- LÓGICA DEL VIDEO DE INTRODUCCIÓN ---
  const introScreen = document.getElementById('intro-screen');
  const introVideo = document.getElementById('intro-video');
  const textElement = document.getElementById('text');

  const iniciarPagina = () => {
    introScreen.classList.add('ocultar-intro');
    document.body.classList.add('scroll-activo');
    textElement.classList.add('animar');
  };

  introVideo.addEventListener('ended', iniciarPagina);

  setTimeout(() => {
    if (!introScreen.classList.contains('ocultar-intro')) {
      iniciarPagina();
    }
  }, 4000);

  // --- LÓGICA DEL SCROLL ---
  const left = document.getElementById("leftImg");
  const right = document.getElementById("rightImg");
  const topImgDesktop = document.getElementById("topImgDesktop");
  const topM = document.getElementById("topImgMobile");
  const bottomM = document.getElementById("bottomImgMobile");
  const finalImgMobile = document.getElementById("finalImgMobile");

  let isTicking = false;

  const updateAnimation = () => {
    const scroll = window.scrollY;
    const max = window.innerHeight * 1.5; 
    let progress = Math.min(scroll / max, 1);

    const movePercentage = progress * 100;

    // Desktop
    left.style.transform = `translateX(${(-100 + movePercentage)}%)`;
    right.style.transform = `translateX(${(100 - movePercentage)}%)`;

    // Móvil
    topM.style.transform = `translateY(${(-100 + movePercentage)}%) scale(1.15)`;
    bottomM.style.transform = `translateY(${(100 - movePercentage)}%) scale(1.15)`;

    // Texto
    let textProgress = Math.min(progress * 1.3, 1);
    const textYOffset = textProgress * 300;
    
    textElement.style.transform = `translate(-50%, calc(-50% - ${textYOffset}px))`;
    textElement.style.opacity = 1 - textProgress;

    if (textProgress >= 0.95) {
      textElement.style.visibility = "hidden";
    } else {
      textElement.style.visibility = "visible";
    }

    // Imágenes finales
    let imgProgress = Math.max((progress - 0.5) * 2, 0);
    topImgDesktop.style.opacity = Math.min(imgProgress, 1);
    finalImgMobile.style.opacity = Math.min(imgProgress, 1);

    isTicking = false;
  };

  window.addEventListener("scroll", () => {
    if (!isTicking) {
      window.requestAnimationFrame(updateAnimation);
      isTicking = true;
    }
  }, { passive: true });
</script>
